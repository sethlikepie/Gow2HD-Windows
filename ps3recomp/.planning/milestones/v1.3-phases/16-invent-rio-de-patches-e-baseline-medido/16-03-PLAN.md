1. Trabalhar no checkout de build:
   ```bash
   cd ../gow2-recomp
   test -x ./boot_gow2
   ```

2. Receita canónica (adaptada de CLAUDE.md — **uma** corrida, não seis):
   ```bash
   LOG=/tmp/m0_fase16_smoke.log
   (
     set -a; . ./env_gow2.sh; set +a
     # limpar traces ruidosos se existirem
     unset $(env | awk -F= '/^PS3_TRACE_/ {print $1}') 2>/dev/null || true
     export PS3_NO_RSX=1 PS3_PERF_FSM=1 PS3_MOVIE_EOS=0
     unset PS3_MOVIE_DONE_MS PS3_VDEC_FORCE_SEQDONE_MS 2>/dev/null || true
     # probes OFF por default — não ligar PS3_*_PROBE
     exec ./boot_gow2 EBOOT.ELF
   ) &gt; "$LOG" 2&gt;&amp;1 &amp;
   BPID=$!
   sleep 25
   kill -TERM $BPID 2&gt;/dev/null
   sleep 1
   kill -9 $BPID 2&gt;/dev/null
   wait $BPID 2&gt;/dev/null || true
   ```

3. **NUNCA** `pkill -f boot_gow2` (CLAUDE.md / G6 — load 75 no host).

4. Extrair métricas do log:
   ```bash
   # st620 max (lado direito do ->), ver smoke_m0_baseline.sh
   ST620_MAX=$(grep -oE '\[MOVIEFSM\] st620 [0-9]+ -&gt; [0-9]+' "$LOG" \
     | awk '{print $NF}' | sort -n | tail -1)
   # sticky (padrões já usados no projecto)
   STICKY=$(grep -cE 'sticky|ps3_fios_sticky' "$LOG" || true)
   # OOB pós-TLS: presença de OOB em faixa 0xFFFF* logo após mensagens TLS, se houver
   OOB=$(grep -cE 'OOB|0xFFFF' "$LOG" || true)
   # órfãos de load se o log tiver o padrão do projecto
   grep -E '\[MOVIEFSM\] st620|sticky|002B4274|OOB' "$LOG" | head -30
   ```

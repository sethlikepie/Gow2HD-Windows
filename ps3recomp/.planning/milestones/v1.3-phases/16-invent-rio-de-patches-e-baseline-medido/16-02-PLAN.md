from pathlib import Path
import re, sys
p = Path("games/gow2/lift_baseline/PATCH_MIGRATION.tsv")

lines = [ln for ln in text.splitlines() if ln and not ln.startswith("#")]

    assert c in idx, f"coluna em falta: {c}"

classified = []
